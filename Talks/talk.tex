\begin{frame}{The Bellman equation}
  $w_{n,p}$ = optimal winning probability with $n$ coins, $w_{0,p}:=1$.

  \medskip
  \[
    \boxed{\;
      w_{n,p}\;=\;p^{n}\;+\;\sum_{j=1}^{n-1}\binom{n}{j}\,p^{n-j}q^{j}\,
      \max_{j\le m\le n-1} w_{m,p}\;}
  \]

  \medskip
  \begin{itemize}
    \item $p^{n}$: all heads $\Rightarrow$ immediate win
    \item $j$ tails: choose remaining $m\in\{j,\dots,n-1\}$
    \item player maximises $w_{m,p}$
  \end{itemize}
\end{frame}

\begin{frame}{The fair case $p=\tfrac12$}
  \begin{block}{Theorem}
    $w_{n,1/2}=\tfrac12$ for every $n\ge 1$.
  \end{block}

  More: every policy taking the immediate win at $k=n$ has
  $w_{n,1/2}=\tfrac12$.

  \medskip
  \textbf{Proof.}\enspace Strong induction on $n$; key identity
  $\sum_{j=1}^{n-1}\binom{n}{j}2^{-n}=1-2^{1-n}$:
  \[
    w_{n,1/2}
    \;=\; 2^{-n} + \tfrac12\bigl(1-2^{1-n}\bigr)
    \;=\; \tfrac12. \qquad \square
  \]

  \medskip
  \begin{itemize}
    \item symmetry axis
    \item all reasonable strategies optimal
    % \item $p>\tfrac12$ vs $p<\tfrac12$: very different worlds
  \end{itemize}
\end{frame}

\begin{frame}{Above: $p>\tfrac12$}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \begin{block}{Theorem}
      For $p>\tfrac12$, $n\ge 2$:
      \begin{itemize}
        \item[(i)] $w_{n,p}>w_{n-1,p}$
        \item[(ii)] \One{} optimal: $w_{n,p}=a_{n,p}$
      \end{itemize}
      $\Rightarrow$ $W(p):=\lim_n w_{n,p}$ exists,
      $\;p\le W(p)<1$,
      \[
        W(p) = \sum_{k\ge 1}p^{k} \prod_{j>k} \bigl(1-p^{j}-q^{j}\bigr).
      \]
    \end{block}

    \medskip
    {\scriptsize\textbf{Proof idea:} joint induction with
    $w_{n-1,p}<\dfrac{p^{n}}{p^{n}+q^{n}}$.}

\begin{frame}{First-order asymptotics: $c_n$}
  Deficit $\Delta_{n,p}:=\tfrac12-w_{n,p}$,\enspace $p=\tfrac12-\delta$:
  \[
    \Delta_{n,p}\;=\;c_{n}\,\delta+O(\delta^{2}),\quad c_{1}=1,
  \]
  \[
    \boxed{\;
      c_{n}\;=\;\frac{n}{2^{n-1}}+\frac{1}{2^{n}}\sum_{j=1}^{n-1}\binom{n}{j}\,
      \min_{j\le m\le n-1}c_{m}\;}\;\; (n\ge 2).
  \]

  \medskip
  \begin{columns}[T]

    \scriptsize
    \begin{align*}

\begin{frame}{Shape of $w_{n,p}$ at first order}
  \begin{block}{Corollary ($p=\tfrac12-\delta$, $\delta>0$ small)}
    \begin{itemize}
      \item[(i)] $w_{n-1,p}-w_{n,p}=(c_{n}-c_{n-1})\,\delta+O(\delta^{2})$
      \item[(ii)] strict local \emph{minimum} at $n=5$
      \item[(iii)] no local maximum at first order
    \end{itemize}
  \end{block}

  \medskip
  \textbf{Proof idea.}\enspace Joint induction on $n$ over four
  statements:
  \begin{itemize}
    \item[(a)] \emph{collapse:} $\min_{j\le m\le n-1}c_{m}=
          \begin{cases}c_{j} & j\le 3\\ c_{n-1} & j\ge 4\end{cases}$
          \quad ($n\ge 7$)
    \item[(b)] \emph{lower bound:} $c_{n}\ge\tfrac{27}{16}$ \quad ($n\ge 4$)
    \item[(c)] \emph{strict decrease:} $c_{n}<c_{n-1}$ \quad ($n\ge 6$)
    \item[(d)] \emph{linear recursion:} $c_{n}=A_{n}+(1-B_{n})\,c_{n-1}$
          \quad ($n\ge 7$)
  \end{itemize}

\end{frame}
